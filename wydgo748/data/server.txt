# WYD-Go - operational configuration
#
# The public IP is not defined here. The server listens on all interfaces;
# the client server list must point to the public server IP.
listen_address=0.0.0.0:8281

# Authoritative persistence. Keep the URL in the environment; never commit
# operational credentials in this file. For local development, set:
#   $env:WYD_DATABASE_URL='postgres://wydgo:<senha>@127.0.0.1:5432/wydgo?sslmode=disable'
database_driver=postgres
database_url_env=WYD_DATABASE_URL
database_max_conns=8
# Nonblocking gameplay persistence: workers must stay below database_max_conns.
critical_persistence_workers=4
critical_persistence_capacity=1024

# Additional PIN required only by the web administration panel. Configure a
# local value of 6 to 12 digits; never commit the production PIN.
admin_access_pin=123456

# Local panel started by tm.exe; logs remain in this console.
# Set false to disable. HTTP is allowed only on a literal loopback IP.
web_admin_enabled=true
web_admin_address=127.0.0.1:8082
web_admin_staff=data/staff.json
web_admin_static=web/portal/dist

# Operational network/authentication protections. The load test overrides
# these settings because hundreds of bots share the same local IP.
session_queue_capacity=256
world_command_queue_capacity=1024
auth_hash_concurrency=4
# Production: at most 400 sockets per channel and four concurrent connections
# per source. The authenticated limit enforces the same barrier after login;
# duplicate logins for the same account remain blocked.
max_connections=400
max_connections_per_ip=4
max_authenticated_clients_per_ip=4
handshake_timeout_seconds=5
session_idle_timeout_seconds=600
frame_read_timeout_seconds=10
inbound_packets_per_second=256
inbound_bytes_per_second=524288
auth_attempts_per_minute_ip=30
auth_attempts_per_minute_account=10
chat_local_per_10_seconds=8
chat_whisper_per_10_seconds=8
chat_global_per_10_seconds=2
critical_persistence_timeout_ms=500
channel_id=1
# Local CIDR allow/deny/limit rules. Login never queries an external API.
network_admission=data/network_admission.json
client_integrity=data/client_integrity.json

# Generator logging: quiet, summary or verbose.
npcgener_log=summary
# Combat/loot diagnostics: summary avoids a log write per hit.
# Use verbose only for a short reproduction; quiet also disables the summary.
gameplay_log=summary

# Boss directory (one .lua per encounter). Bosses do not use NPCGener:
# each file defines its own position and respawn. See DOCS/server/operations.md.
boss=data/boss

# Diagnostics: metrics at /debug/vars and profiling at /debug/pprof.
# Commented out means disabled (default). The host must be loopback: these
# endpoints expose internal state and can trigger profiling. A public address
# deliberately fails startup. For remote access, use an SSH tunnel:
#   ssh -L 6060:127.0.0.1:6060 user@server
# debug_address=127.0.0.1:6060

# Global balancing.
# exp_rate is a percentage: 100=normal, 200=double, 50=half.
# Apply the floor first, then the rate and party bonus.
# exp_rate=0 disables EXP gain.
exp_rate=100
exp_minimum=10000
party_exp_bonus=2

# Compositor (Anct). Composition requires exactly four materials, all from
# set D (item level 4) or E (item level 5), all refined to +7, +8 or +9.
# Success chance in percent is compositor_base plus each material's bonus,
# selected by its refine level. The values below cap the chance at 10+8*4=42.
#   4 materials +7 = 22   4 materials +8 = 30   4 materials +9 = 42
compositor_base=10
compositor_refine_7=3
compositor_refine_8=5
compositor_refine_9=8

# Permanent world objects (gates, doors, cannons, towers). Added to the map at
# startup; they neither expire nor allow pickup.
init_items=data/init_items.csv

# Authoritative data.
npcs=data/npcs
npcgener=data/NPCGener.txt
teleports=data/teleports.ini
# Used only when database_driver=json.
accounts=data/accounts
# Guild registry. Keep it outside the account directory; account scanners
# would otherwise attempt to parse it as an account.
guilds=data/guilds.json
# Native Guilds.txt for distribution to clients: WYD.exe 7.48 reads
# "./Guilds.txt" from its own directory. Without it, guild names do not appear
# (7.48 has no guild-name packet). Copy the generated file into the client
# directory. Format: <group> <channel> <index> <name>.
guilds_txt=data/Guilds.txt
# Quests. NPCs absent from this allowlist never become quests.
quests=data/quests.json
items=data/itemlist.csv
itemnames=data/Itemname.csv
itemeffects=data/ItemEffect.h
skills=data/SkillData.csv
droprates=data/droprate.json
volatiles=data/volatiles.json
instances=data/instances.json
repliction=data/repliction.json
mounts=data/mounts.json
character_templates=data/character_templates.json

heightmap=data/maps/HeightMap.dat
attributemap=data/maps/AttributeMap.dat
