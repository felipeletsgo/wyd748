// Decrypt decifra o pacote in-place (do byte 4 ate size). Retorna (ok checksum).
// Algoritmo identico ao CPSock::ProcessMsg do TMSrv 759.
func Decrypt(msg []byte) bool {
	size := int(uint16(msg[0]) | uint16(msg[1])<<8)
	if size < 4 || size > len(msg) {
		return false
	}
	iKey := msg[2]
	checkSum := msg[3]
	key := pKeyWord[int(iKey)*2]

	var sum1, sum2 byte
	pos := int(key)
	for i := 4; i < size; i, pos = i+1, pos+1 {
		sum2 += msg[i]
		trans := pKeyWord[(pos%256)*2+1]
		switch i & 3 {
		case 0:
			msg[i] -= trans << 1
		case 1:
			msg[i] += trans >> 3
		case 2:
			msg[i] -= trans << 2
		case 3:
			msg[i] += trans >> 5
		}
		sum1 += msg[i]
	}
	return sum2-sum1 == checkSum
}

// Encrypt cifra o pacote in-place (inverso do Decrypt) e escreve KeyWord/CheckSum.
func Encrypt(msg []byte, iKey byte) {
	size := int(uint16(msg[0]) | uint16(msg[1])<<8)
	if size < 4 || size > len(msg) {
		return
	}
	msg[2] = iKey
	key := pKeyWord[int(iKey)*2]

	var sum1, sum2 byte
	pos := int(key)
	for i := 4; i < size; i, pos = i+1, pos+1 {
		sum1 += msg[i]
		trans := pKeyWord[(pos%256)*2+1]
		switch i & 3 {
		case 0:
			msg[i] += trans << 1
		case 1:
			msg[i] -= trans >> 3
		case 2:
			msg[i] += trans << 2
		case 3:
			msg[i] -= trans >> 5
		}
		sum2 += msg[i]
	}
	msg[3] = sum2 - sum1
}
